function trigramTopToBottom(key){
  return TRIGRAMS[key].bottomToTop.slice().reverse();
}

function buildYaoLineEl(isYang){
  const row = document.createElement('div');
  row.className = 'yao-line ' + (isYang ? 'yang' : 'yin');
  if(isYang){
    const bar = document.createElement('div');
    bar.className = 'bar';
    row.appendChild(bar);
  } else {
    const bar1 = document.createElement('div');
    bar1.className = 'bar';
    const bar2 = document.createElement('div');
    bar2.className = 'bar';
    row.appendChild(bar1);
    row.appendChild(bar2);
  }
  return row;
}

function renderStaticHexagram(container, upperKey, lowerKey){
  container.innerHTML = '';

  seq.forEach(bit=>{

  });
}

    setTimeout(()=>{
      els[idx].classList.add('drawn');
      if(i === order.length - 1){
        setTimeout(onDone, 350);
      }
    }, i * 260);
  });
}

function buildCompass(hex){
  const svg = document.getElementById('compassSvg');
  const NS = 'http://www.w3.org/2000/svg';
  svg.innerHTML = '';
  const cx = 160, cy = 160, rOuter = 140, rRing = 118, rLabel = 92, rGlyph = 62;

  const ring = document.createElementNS(NS, 'circle');
  ring.setAttribute('class','ring');
  ring.setAttribute('cx', cx); ring.setAttribute('cy', cy); ring.setAttribute('r', rRing);
  svg.appendChild(ring);

  const centerRing = document.createElementNS(NS, 'circle');
  centerRing.setAttribute('class','ring');
  centerRing.setAttribute('cx', cx); centerRing.setAttribute('cy', cy); centerRing.setAttribute('r', 34);
  svg.appendChild(centerRing);

  Object.keys(TRIGRAMS).forEach(key=>{
    const t = TRIGRAMS[key];
    const rad = (t.bearing - 90) * Math.PI / 180; // -90 使 0 度朝上
    const x1 = cx + 34 * Math.cos(rad), y1 = cy + 34 * Math.sin(rad);
    const x2 = cx + rRing * Math.cos(rad), y2 = cy + rRing * Math.sin(rad);
    const isActive = (key === hex.u || key === hex.l);

    const spoke = document.createElementNS(NS,'line');
    spoke.setAttribute('class', 'spoke' + (isActive ? ' active' : ''));
    spoke.setAttribute('x1', x1); spoke.setAttribute('y1', y1);
    spoke.setAttribute('x2', x2); spoke.setAttribute('y2', y2);
    svg.appendChild(spoke);

    const gx = cx + rGlyph * Math.cos(rad), gy = cy + rGlyph * Math.sin(rad);
    const glyphText = document.createElementNS(NS,'text');
    glyphText.setAttribute('x', gx); glyphText.setAttribute('y', gy + 6);
    glyphText.setAttribute('text-anchor','middle');
    glyphText.setAttribute('font-size', isActive ? '22' : '18');
    glyphText.setAttribute('fill', isActive ? 'var(--seal-deep)' : 'var(--ink-soft)');
    glyphText.textContent = t.glyph;
    svg.appendChild(glyphText);

    const nx = cx + rLabel * Math.cos(rad), ny = cy + rLabel * Math.sin(rad);
    const g = document.createElementNS(NS,'g');
    g.setAttribute('class','hit' + (isActive ? ' active' : ''));

    const nameText = document.createElementNS(NS,'text');
    nameText.setAttribute('class','name-label');
    nameText.setAttribute('x', nx); nameText.setAttribute('y', ny - 4);
    nameText.setAttribute('text-anchor','middle');
    nameText.textContent = key;
    g.appendChild(nameText);

    const dirText = document.createElementNS(NS,'text');
    dirText.setAttribute('class','dir-label');
    dirText.setAttribute('x', nx); dirText.setAttribute('y', ny + 14);
    dirText.setAttribute('text-anchor','middle');
    dirText.textContent = t.dir;
    g.appendChild(dirText);

    svg.appendChild(g);
  });
}

function buildYaoBlocks(yaoList){
  const container = document.getElementById('yaoContainer');
  container.innerHTML = '';

  yaoList.forEach((row, idx)=>{
    const [label, text, textX, xiangText, xiangX] = row;

    const block = document.createElement('div');
    block.className = 'yao-block';

    const head = document.createElement('div');
    head.className = 'yao-head';
    const labelEl = document.createElement('span');
    labelEl.className = 'yao-label';
    labelEl.textContent = label;
    const textEl = document.createElement('p');
    textEl.className = 'yao-text serif';
    textEl.textContent = text;
    head.appendChild(labelEl);
    head.appendChild(textEl);

    const explainEl = document.createElement('p');
    explainEl.className = 'yao-explain';
    explainEl.textContent = textX;

    const toggleBtn = document.createElement('button');
    toggleBtn.className = 'yao-toggle';
    toggleBtn.type = 'button';
    toggleBtn.textContent = '象曰';
    toggleBtn.setAttribute('aria-expanded', 'false');

    const panel = document.createElement('div');
    panel.className = 'yao-xiang-panel';
    panel.hidden = true;

    const panelLabel = document.createElement('p');
    panelLabel.className = 'classic-label';
    panelLabel.textContent = '小象';
    const panelText = document.createElement('p');
    panelText.className = 'classic-text serif';
    panelText.textContent = xiangText;
    const panelExplain = document.createElement('p');
    panelExplain.className = 'classic-explain';
    panelExplain.textContent = xiangX;
    panel.appendChild(panelLabel);
    panel.appendChild(panelText);
    panel.appendChild(panelExplain);

    toggleBtn.addEventListener('click', ()=>{
      const isHidden = panel.hidden;
      panel.hidden = !isHidden;
      toggleBtn.setAttribute('aria-expanded', String(isHidden));
    });

    block.appendChild(head);
    block.appendChild(explainEl);
    block.appendChild(toggleBtn);
    block.appendChild(panel);
    container.appendChild(block);
  });
}
